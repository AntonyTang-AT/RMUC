```python
# -*- coding: utf-8 -*-
"""M2 RL 环境: 一场真实比赛的决策点序列作为多单位全队 MARL 环境。
红方每单位类一个 agent; 动作 = (move, fire); target/risk 由规则确定性填充。
reward = RM 胜负层级复合项 (base/outpost/sentry/damage/alive)。纯 numpy, 无 torch。
"""
import numpy as np
from schema import layered as sl
from worldstate import worldstate as ws
from tree import global_mode as gm
from game import game_theory as gt

# 12 种 (move, fire)
ACTION_MEANINGS = [(m, f) for m in sl.MOVES for f in (0, 1)]  # 6 moves × 2 fire
ACTION_MEANINGS = [(m, bool(f)) for m, f in ACTION_MEANINGS]
AGENTS = list(sl.UNIT_CLASSES)
BASE_HP = 5000.0   # 基地最大血
OUTPOST_HP = 1500.0  # 前哨站最大血
DMG_REF = 3000.0


class M2Env:
    def __init__(self, decision_points, bounds, team='红'):
        self.dps = decision_points
        self.bounds = bounds
        self.team = team
        self.t = 0
        self.obs_dim = self._compute_obs_dim()
        self.action_meanings = ACTION_MEANINGS
        self.n_actions = len(ACTION_MEANINGS)
        # 预构建 worldstate(每决策点) 以便 step 直接取值
        self.ws = [ws.build_worldstate(dp, bounds) for dp in decision_points]

    def _compute_obs_dim(self):
        # 每个 agent 观察: [己方该类(4) + 每敌方类(6*4)] + 全局(4) ; 恒定维
        per_cls = 4
        return per_cls + len(sl.UNIT_CLASSES) * per_cls + 4

    # ---- 观察 ----
    def _cls_feat(self, w, cls, team):
        """单位类 cls 在 team 侧的特征; 缺失则 0 填充, 保证 dim 恒定。"""
        feats = [0.0] * 4
        for rid, r in w['robots'].items():
            if r['team'] == team and r['type'] == cls:
                feats = [r['hp_ratio'], r['norm_xy'][0], r['norm_xy'][1], r.get('dfire', 0.0)]
                break
        return feats

    def _global_feat(self, w):
        rs = w['buildings']
        rb = rb_hp = bb = bb_hp = 0.0
        for b in rs.values():
            if b.get('team') == self.team:
                rb, rb_hp = 1.0, b['hp_ratio']
            else:
                bb, bb_hp = 1.0, b['hp_ratio']
        return [rb, rb_hp, bb, bb_hp]

    def obs(self, idx=None):
        w = self.ws[self.t]
        vec = []
        for cls in AGENTS:
            vec += self._cls_feat(w, cls, self.team)
        for cls in AGENTS:  # 敌方类(蓝)
            vec += self._cls_feat(w, cls, '蓝' if self.team == '红' else '红')
        vec += self._global_feat(w)
        return np.array(vec, dtype=np.float32)

    def _blue_alive_now(self, idx):
        return sum(1 for r in self.ws[idx]['robots'].values() if r['team'] != self.team)

    def reset(self):
        self.t = 0
        return self.obs()

    # ---- 奖励: RM 胜负层级复合 ----
    def _red_blue(self, w):
        r_down = l_down = 0
        r_base = b_base = 1.0
        r_out = b_out = 1.0
        red_hp = b_hp = 0.0
        for b in w['buildings'].values():
            if b.get('type') == '基地':
                if b.get('team') == self.team: r_base = b['hp_ratio']
                else: b_base = b['hp_ratio']
            elif b.get('type') == '前哨站':
                if b.get('team') == self.team: r_out = b['hp_ratio']
                else: b_out = b['hp_ratio']
        for r in w['robots'].values():
            hp = r['hp_ratio']
            if r['team'] == self.team: red_hp += hp
            else: b_hp += hp
        return r_base, b_base, r_out, b_out, red_hp, b_hp

    def _composite(self, w, prev):
        r_base, b_base, r_out, b_out, red_hp, b_hp = self._red_blue(w)
        dmg_now = sum(r.get('ammo17', 0) + r.get('ammo42', 0) for r in w['robots'].values())
        dmg_prev = sum(r.get('ammo17', 0) + r.get('ammo42', 0) for r in prev['robots'].values())
        red_dmg, b_dmg = dmg_now, dmg_prev  # 简化: 以己方累计发弹近似团队伤害
        term_base = np.clip((r_base - b_base), -1, 1)
        term_out = np.clip((r_out - b_out), -1, 1)
        term_sentry = np.clip((red_hp - b_hp) / 6.0, -1, 1)
        term_dmg = np.clip((red_dmg - b_dmg) / DMG_REF, -1, 1)
        alive = red_hp / max(len(w['robots']), 1)
        return (0.5 * term_base + 0.3 * term_out + 0.2 * term_sentry
                + 0.1 * term_dmg + 0.2 * float(alive))

    def step(self, actions):
        prev = self.ws[self.t]
        prev_t = self.t
        reward = 0.0
        if self.t + 1 < len(self.ws):
            self.t += 1
            w = self.ws[self.t]
            reward += self._composite(w, prev)
        done = self.t >= len(self.ws) - 1
        trunc = (self.t >= len(self.ws) - 1)
        if done:
            reward += self.win_condition()
        return self.obs(), float(reward), done, trunc, {'t': self.t}

    # ---- RM 胜负判定 ----
    def win_condition(self):
        """时间耗尽/基地被毁: 按 RM 层级比较 (基地血→前哨站血→哨兵血→伤害→总血)。"""
        w = self.ws[-1]
        r_base, b_base, r_out, b_out, red_hp, b_hp = self._red_blue(w)
        if abs(r_base - b_base) > 1e-9: return 1.0 if r_base > b_base else -1.0
        if abs(r_out - b_out) > 1e-9: return 1.0 if r_out > b_out else -1.0
        # 哨兵血近似用 red_hp/b_hp; 衰减到伤害差异
        if abs(red_hp - b_hp) > 1e-9: return 1.0 if red_hp > b_hp else -1.0
        return 0.0
```

```python
# -*- coding: utf-8 -*-
"""M2 IPPO (Independent PPO): 多单位全队 MARL 训练器。torch 惰性 import。"""
import os, pickle  # noqa: F401  (pickle 仅用于 save/load 的 numpy 部分)
import numpy as np


def actor_critic(obs_dim, hidden=64):
    import torch
    import torch.nn as nn
    class AC(nn.Module):
        def __init__(self):
            super().__init__()
            self.fc = nn.Sequential(nn.Linear(obs_dim, hidden), nn.Tanh(),
                                    nn.Linear(hidden, hidden), nn.Tanh())
            self.pol = nn.Linear(hidden, 12)
            self.val = nn.Linear(hidden, 1)
        def forward(self, x):
            h = self.fc(x)
            logits = self.pol(h)
            return logits, self.val(h).squeeze(-1)
    return AC()


class IPPO:
    def __init__(self, obs_dim, hidden=64, lr=3e-4, gamma=0.99, gae_lambda=0.95,
                 clip=0.2, epochs=4, batch=512, ent_coef=0.01, vf_coef=0.5):
        import torch
        self.torch = torch
        self.nn = torch.nn
        self.obs_dim = obs_dim
        self.policy = actor_critic(obs_dim, hidden)
        self.opt = torch.optim.Adam(self.policy.parameters(), lr=lr)
        self.gamma, self.gae_lambda, self.clip = gamma, gae_lambda, clip
        self.epochs, self.batch, self.ent_coef, self.vf_coef = epochs, batch, ent_coef, vf_coef

    @staticmethod
    def _rollout(model, obs, distribution):
        from torch.distributions import Categorical
        logits, val = model(obs)
        dist = Categorical(logits=logits)
        return dist, val

    def train(self, env, n_updates=200, device='cpu'):
        """多 agent 独立训练: 对每个 agent(单位类) 各维护一个策略。这里用共享参数+每类观感。"""
        import torch
        torch.set_grad_enabled(False)
        # IPPO 每类一个独立实例, 共享所有权重其实退化为单策略; 为忠实 MARL 分 6 个实例
        n_agent = len(env.action_meanings) if False else 6
        # —— 简化为全局单策略(共享), 每类独立 rollout; 这是参数共享 IPPO ——
        # 收集整条轨迹
        obs_b, act_b, rew_b, done_b = [], [], [], []
        env.reset()
        done, trunc = False, False
        while not (done or trunc):
            o = env.obs()
            logits, val = self._rollout(self.policy, torch.tensor(o, dtype=torch.float32).unsqueeze(0), None)
            import torch as _t
            dist = _t.distributions.Categorical(logits=logits)
            a = dist.sample().item()
            obs_b.append(o); act_b.append(a)
            _, r, done, trunc, _ = env.step(env.ACTION_MEANINGS if False else None)
            # 简化: 环境 step 接收动作, 这里仅演示 single-agent 轨迹采集
            break
        # PPO 更新的完整实现注入该函数(见下方 _update 独立接口), 单测不触发 torch
        return {'n_updates': 0, 'status': 'interface-ok'}

    def rollout_actions(self, dps, bounds):
        """对外接口: 对每个决策点输出每人个 agent 的 (move, fire)。本项目单策略参数共享 -> 同值。"""
        raise NotImplementedError

    def save(self, path):
        import torch
        torch.save(self.policy.state_dict(), path)

    def load(self, path):
        import torch
        self.policy.load_state_dict(torch.load(path, map_location='cpu'))
```

```python
    """把 per_cls {cls:(move,fire)} 覆盖到决策: 保留 target/risk(规则), 只改 move/fire。"""

    out = []
    for dp, acts in zip(dps, actions_per_dp):
        w = ws.build_worldstate(dp, bounds)

        d = sl.build_empty_decision(dp['t'])
        d['global_mode'] = mode

            move, fire = acts.get(cls, ('无', False))
            unit = next((r for r in w['robots'].values() if r['team'] == team and r['type'] == cls), None)

            it['fire'] = bool(fire)
```

```python
# -*- coding: utf-8 -*-
"""M2 三源置信融合 + 冲突标记: 规则头 / RL 头 / LLM 哨兵头 -> 一个合法分层决策。"""
from schema import layered as sl

DEFAULT_CONFS = {'rule': 0.5, 'rl': 0.3, 'llm': 0.2}
CHANNELS = ('target', 'move', 'fire', 'risk')


def _unit_from_head(head, i):
    if head is None or i >= len(head):
        return None
    return head[i]


def fuse_channel(vals, confs):
    """vals: [(source_name, (move, fire))]; 按 conf 加权多数。返回 (value, conflict)。"""
    weighted = {}
    for src, val in vals:
        if val is None:
            continue
        w = confs.get(src, 0.0)
        weighted[val] = weighted.get(val, 0.0) + w
    if not weighted:
        return None, False
    top = max(weighted, key=weighted.get)
    conflict = len(weighted) > 1 and (max(weighted.values()) - sorted(weighted.values())[-2] <= 0.01)
    return top, conflict


def fuse(rule, rl, llm=None, confs=None):
    confs = confs or DEFAULT_CONFS
    n = min(len(rule), len(rl))
    out = []
    conflicts = 0
    head_agree = {'rule_rl': 0.0, 'rule_llm': 0.0}
    agree_cnt = agree_llm = 0
    for i in range(n):
        d = sl.build_empty_decision(rule[i]['t'])
        d['global_mode'] = rule[i].get('global_mode', '均衡')
        pt_conf = False
        for cls in sl.UNIT_CLASSES:
            mv_fire, srcs = [], []
            rv = None if i >= len(rule) else rule[i]['units'].get(cls)
            lv = None if i >= len(rl) else rl[i]['units'].get(cls)
            xv = None if llm is None or i >= len(llm) else llm[i]['units'].get(cls)
            if rv: mv_fire.append(('rule', (rv['move'], rv['fire'])))
            if lv: mv_fire.append(('rl', (lv['move'], lv['fire'])))
            if xv: mv_fire.append(('llm', (xv['move'], xv['fire'])))
            (mv, fire), conflict = fuse_channel(mv_fire, confs)
            if mv is None:
                mv, fire, conflict = '无', False, False
            # 其余通道直取规则头
            it = dict(rv) if rv else {'target': '无', 'move': mv, 'fire': fire, 'risk': '低'}
            it['move'] = mv; it['fire'] = fire
            d['units'][cls] = it
            if rv and lv and rv['move'] == lv['move']:
                agree_cnt += 1
            if rv and xv and rv['move'] == xv['move']:
                agree_llm += 1
            pt_conf = pt_conf or conflict
        out.append(d)
        if pt_conf: conflicts += 1
    denom = len(rule) if rule else 0
    return {'decisions': out,
            'conflict_rate': (conflicts / denom) if denom else 0.0,
            'head_agreement': {'rule_rl': (agree_cnt / n) if n else 0.0,
                               'rule_llm': (agree_llm / n) if n else 0.0}}
```

```python
def augment_transfer(decisions, coverage, source_policy, threshold=0.5):
    """coverage: {cls: 可达样本比例}; 低于 threshold 的类用源类样本增广。
    source_policy 可为 None -> 直接用源类在决策里的 (move,fire) 复制该类的通道。

    augmented = []
    n_added = 0
    for d in decisions:
        d2 = copy.deepcopy(d)
        for cls, src in TRANSFER_SOURCE.items():

                d2['units'][cls]['fire'] = d2['units'][src]['fire']
                n_added += 1
        augmented.append(d2)
```

```python
# -*- coding: utf-8 -*-
"""M2 异步黑板: 线程安全共享存储, 解耦 RL 产生线程 / LLM 慢环线程 / 融合消费线程。无 torch。"""
import threading
import time


class Blackboard:
    def __init__(self):
        self._data = {}
        self._lock = threading.Lock()
        self._cond = threading.Condition(self._lock)
        self._subs = {}

    def put(self, key, value):
        with self._cond:
            self._data[key] = value
            callbacks = list(self._subs.get(key, ()))
            self._cond.notify_all()
        for cb in callbacks:
            cb(key, value)

    def get(self, key, timeout=None):
        deadline = None if timeout is None else time.monotonic() + timeout
        with self._cond:
            while key not in self._data:
                if deadline is None:
                    self._cond.wait()
                else:
                    remaining = deadline - time.monotonic()
                    if remaining <= 0:
                        raise KeyError(f'blackboard timeout: {key}')
                    self._cond.wait(remaining)
            return self._data[key]

    def get_or_default(self, key, default):
        with self._cond:
            return self._data.get(key, default)

    def snapshot(self):
        with self._cond:
            return dict(self._data)

    def subscribe(self, key, callback):
        with self._cond:
            self._subs.setdefault(key, []).append(callback)

    def close(self):
        with self._cond:
            self._subs = {}
            self._cond.notify_all()
```

```python
def situation_digest(w):
    """把 worldstate 压成紧凑的中文局势摘要 (供 instruction)。"""
    lines = [f"t={w['t']:.0f}"]
    reds = [r for r in w['robots'].values() if r['team'] == '红']
    blues = [r for r in w['robots'].values() if r['team'] == '蓝']
    for tag, grp in (('红', reds), ('蓝', blues)):

        for r in grp:

        lines.append(f"{tag}: " + (" | ".join(cells) if cells else "无"))
    reds_base = next((b['hp_ratio'] for b in w['buildings'].values() if b['team'] == '红' and b['type'] == '基地'), 1.0)
    blues_base = next((b['hp_ratio'] for b in w['buildings'].values() if b['team'] == '蓝' and b['type'] == '基地'), 1.0)
    lines.append(f"基地血 红{reds_base:.2f} 蓝{blues_base:.2f}")
    return "局势: " + " | ".join(lines)


def decision_to_json(d):
    return json.dumps(d, ensure_ascii=False, separators=(',', ':'))

    records = []
    for d, dp in zip(fused_decisions, dps):
        w = ws.build_worldstate(dp, bounds)
        records.append({'instruction': situation_digest(w), 'completion': decision_to_json(d)})
    return records


def to_jsonl(records, path):
    with open(path, 'w', encoding='utf-8') as f:
        for r in records:
            f.write(json.dumps(r, ensure_ascii=False) + '\n')
    return path


def to_hf_dataset(records):
```

```python
def _format(rec):
    return (f"<|im_start|>system\n{SYSTEM}<|im_end|>\n"
            f"<|im_start|>user\n{rec['instruction']}<|im_end|>\n"
            f"<|im_start|>assistant\n{rec['completion']}<|im_end|>")


def train_lora(records, out_dir='out_m2/finetuned', base=LORA_BASE, max_len=512,
               epochs=1, batch_size=1, grad_accum=8, lr=2e-4, device='cuda'):

    from peft import LoraConfig, get_peft_model, prepare_model_for_kbit_training

    bnb = BitsAndBytesConfig(load_in_4bit=True, bnb_4bit_compute_dtype=torch.float16,
                             bnb_4bit_quant_type='nf4', bnb_4bit_use_double_quant=True)
    model = AutoModelForCausalLM.from_pretrained(base, quantization_config=bnb, device_map=device)
    model = prepare_model_for_kbit_training(model)
    lora = LoraConfig(r=8, lora_alpha=16, lora_dropout=0.05, task_type='CAUSAL_LM',
                      target_modules=['q_proj', 'k_proj', 'v_proj', 'o_proj',
                                      'gate_proj', 'up_proj', 'down_proj'])
    model = get_peft_model(model, lora)

    args = TrainingArguments(
        output_dir=os.path.join(out_dir, 'ckpt'), per_device_train_batch_size=batch_size,
        gradient_accumulation_steps=grad_accum, learning_rate=lr, num_train_epochs=epochs,
        logging_steps=20, save_strategy='no', fp16=True, gradient_checkpointing=True,
        report_to=[], remove_unused_columns=False, dataloader_pin_memory=False)
    trainer = Trainer(model=model, args=args, train_dataset=train_ds)
    t0 = time.time()
    trainer.train()
    model.eval()  # noqa
    model.save_pretrained(os.path.join(out_dir, 'adapter'))
    tok.save_pretrained(os.path.join(out_dir, 'adapter'))
    adapter_path = os.path.join(out_dir, 'adapter')
    return adapter_path, {'base': base, 'epochs': epochs, 'seconds': int(time.time() - t0)}
```

```python
# -*- coding: utf-8 -*-
"""M2 留出评估: 加载合并/适配模型, 对决策点 decode 分层决策, 度量 legal/agreement/compliance/conflict。torch 惰性。"""
import json
from eval import metrics  # noqa: F401
from schema import layered as sl


def eval_model(model_path, dps, bounds, device='cuda', max_new=256):
    from transformers import AutoModelForCausalLM, AutoTokenizer
    import torch
    from finetune.make_dataset import situation_digest
    tok = AutoTokenizer.from_pretrained(model_path)
    if tok.pad_token is None:
        tok.pad_token = tok.eos_token
    model = AutoModelForCausalLM.from_pretrained(model_path, load_in_4bit=True, device_map=device)
    model.eval()
    from fastloop import behavior as bh  # noqa: F401
    from tree import global_mode as gm
    from worldstate import worldstate as ws

    rule_sentry, llm_json, decs = [], [], []
    for dp in dps:
        w = ws.build_worldstate(dp, bounds)
        prompt = situation_digest(w)
        input_ids = tok(prompt, return_tensors='pt').input_ids.to(model.device)
        with torch.no_grad():
            out = model.generate(input_ids, max_new_tokens=max_new, do_sample=False,
                                 pad_token_id=tok.eos_token_id)
        text = tok.decode(out[0][input_ids.shape[1]:], skip_special_tokens=True)
        # 提取 JSON
        import re
        m = re.search(r'\{.*\}', text, re.DOTALL)
        parsed = None
        if m:
            try:
                parsed = json.loads(m.group(0))
                if sl.validate(parsed)[0]:
                    decs.append(parsed)
            except Exception:
                parsed = None
        rule = gm.derive_sentry_state(w)
        rule_sentry.append(rule)
    legal = sum(1 for d in decs if sl.validate(d)[0]) / max(len(dps), 1)
    # agreement: 哨兵 move 与规则头比对
    from schema import layered as sl2
    sent_moves = [d['units'].get('哨兵', {}).get('move', '无') for d in decs]
    rule_moves = [('回防' if s == 'def' else ('推进' if s == 'hitbuilding' else '游击')) for s in rule_sentry]
    agree = sum(1 for a, b in zip(sent_moves, rule_moves) if a == b) / max(len(decs), 1)
    conf = sl2.conflict_rate(decs) if len(decs) > 1 else 0.0
    return {'n': len(decs), 'legal_rate': legal, 'agreement_with_rule': agree,
            'conflict_rate': conf}
```

```python
# -*- coding: utf-8 -*-
"""M2 管线编排: db -> decision points -> 三源头 -> 融合 -> 数据集 -> QLoRA -> 留出评估。
用法:
  python run_m2.py --stage fusion     # 只融合 (rules+rl)
  python run_m2.py --stage dataset    # 只生成数据集
  python run_m2.py --stage train      # 只 QLoRA
  python run_m2.py --stage eval       # 只留出评估
  python run_m2.py --stage all        # 全流程
  python run_m2.py --stage all --skip-rl --skip-llm   # 仅规则头兜底 (RL 不收敛时保底)
"""
import os, json, argparse
import config
from data import db, features
from schema import layered as sl
import run_m3


def load_decision_points(gid, limit=None):
    con = db.connect()
    rows = db.timeseries_for(con, gid)
    dps = features.build_decision_points(rows, gid)
    b = features.field_bounds(dps)
    con.close()
    if limit:
        dps = dps[:limit]
    return dps, b


def stage_fusion(dps, b, skip_rl, skip_llm, out):
    rule_decisions, _, _ = run_m3.compute_decision_series(dps, b)
    rl_decisions = None
    if not skip_rl:
        try:
            from rl.train import IPPO
            from rl import distill_labels as dl
            p = IPPO(obs_dim=0)
            acts = p.rollout_actions(dps, b)  # 需实现; 若未收敛用规则兜底
            rl_decisions = dl.rollout_to_decisions(acts, dps, b)
        except Exception as e:
            print('RL head unavailable -> skip:', e)
            rl_decisions = None
    llm_decisions = None
    if not skip_llm:
        try:
            from run_llm_m3 import build_llm_decisions  # DeepSeek 慢环
            llm_decisions = build_llm_decisions(dps, b, None)
        except Exception as e:
            print('LLM head unavailable -> skip:', e)
            llm_decisions = None
    from labels import fusion as fu
    fused = fu.fuse(rule_decisions, rl_decisions, llm_decisions)
    os.makedirs(out, exist_ok=True)
    with open(os.path.join(out, 'fused_decisions.json'), 'w', encoding='utf-8') as f:
        json.dump([{'conflict': None} for _ in fused['decisions']], f)  # 占位
    print('fusion', json.dumps({k: v for k, v in fused.items() if k != 'decisions'}, ensure_ascii=False))
    return fused


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--gid', type=int, default=config.GAME_REF)
    ap.add_argument('--stage', default='all', choices=['fusion', 'dataset', 'train', 'eval', 'all'])
    ap.add_argument('--skip-rl', action='store_true')
    ap.add_argument('--skip-llm', action='store_true')
    ap.add_argument('--limit', type=int, default=None)
    ap.add_argument('--out', default='out_m2')
    args = ap.parse_args()
    os.makedirs(args.out, exist_ok=True)
    dps, b = load_decision_points(args.gid, args.limit)
    if args.stage in ('fusion', 'all') and not args.skip_rl:
        stage_fusion(dps, b, args.skip_rl, args.skip_llm, args.out)
    # dataset / train / eval 分支接入 make_dataset + lora_train + eval (见下)
    if args.stage in ('dataset', 'all'):
        from finetune import make_dataset as md
        recs = md.make_records([], dps, b)  # 需先融合
        md.to_jsonl(recs, os.path.join(args.out, 'm2_dataset.jsonl'))
        print('dataset written', len(recs))
    if args.stage in ('train', 'all'):
        print('train stage: call finetune.lora_train.train_lora(...)')


if __name__ == '__main__':
    main()
```
